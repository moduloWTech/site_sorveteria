 * Ordre : statut/horaires (toujours) â†’ smooth scroll â†’ intro (pendant que la 3D
 * se prÃ©pare) â†’ animations des sections. La 3D n'est jamais abandonnÃ©e : si elle
 * n'est pas prÃªte Ã  la fin de l'intro, elle apparaÃ®t en fondu dÃ¨s qu'elle l'est.

import { initStatus } from 'emg-topas/status';
import { setupScroll } from 'emg-topas/scroll';
import { loadIntro, revealIntro } from 'emg-topas/intro';
import { initSections } from 'emg-topas/sections';
import { initCursor } from 'emg-topas/cursor';
import { preloadAll } from 'emg-topas/preload';
import { initCredit } from 'emg-topas/credit';
import { initMap } from 'emg-topas/vmap';

const root = document.documentElement;
const data = JSON.parse(document.getElementById('emg-topas-data')?.textContent || '{}');
const reduced = matchMedia('(prefers-reduced-motion: reduce)').matches;

/** Ã‰tat partagÃ© entre les animations GSAP (Ã©criture) et la 3D (lecture). */
const state = (window.__topas = {
	reduced,
	hero: { p: 0, mx: 0, my: 0, enter: 0 },
	build: { p: 0 },
	drinks: { p: 0 },
	rain: { active: false, reset: null },
});

function webgl() {
	try {
		const c = document.createElement('canvas');
		return !!(window.WebGL2RenderingContext && c.getContext('webgl2'));
	} catch (e) {
		return false;
	}
}

/** Lance la 3D ; rÃ©essaie une fois en cas d'Ã©chec rÃ©seau ponctuel. */
function loadGL() {
	if (data.no3d || !webgl()) return Promise.resolve(null);
	const attempt = () => import('emg-topas/gl/stage').then((m) => m.createStage(state, data));
	return attempt()
		.catch(() => new Promise((r) => setTimeout(r, 1200)).then(attempt))
		.catch((e) => {
			console.error('[TÃ²pas] 3D indisponible :', e);
			return null;
		});
}

/** Active la 3D (fondu croisÃ© avec les illustrations de secours). */
function showGL(gl, late = false) {
	if (!gl || root.classList.contains('has-gl')) return;
	root.classList.add('has-gl');
	if (window.gsap && !reduced) {
		window.gsap.fromTo('[data-view] canvas', { opacity: 0 }, { opacity: 1, duration: 0.9, ease: 'power2.out' });
		// ArrivÃ©e tardive : le pincho du hero fait son entrÃ©e Ã  ce moment-lÃ .
		if (late) window.gsap.fromTo(state.hero, { enter: 0 }, { enter: 1, duration: 1.8, ease: 'power2.out' });
	} else {
		state.hero.enter = 1;
	}
}

/** Attend que le navigateur ait rÃ©ellement peint (le loader doit apparaÃ®tre en premier). */
const painted = () => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(r)));

async function boot() {
	initStatus(data);
	initCredit();

	const { gsap, ScrollTrigger, SplitText, CustomEase } = window;
	if (reduced || !gsap || !ScrollTrigger) {
		// Statique : pincho complet, verres qui trinquent, pas de pluie.
		root.classList.add('reduced');
		state.build.p = 0.75;
		state.drinks.p = 1;
		state.hero.enter = 1;
		initMap({ reduced: true });
		loadGL().then(showGL);
		return;
	}
	gsap.registerPlugin(ScrollTrigger, SplitText, CustomEase);
	// Mobile : la barre d'adresse qui apparaÃ®t/disparaÃ®t ne doit pas recalculer les pins (sauts).
	ScrollTrigger.config({ ignoreMobileResize: true, autoRefreshEvents: 'visibilitychange,DOMContentLoaded,load' });
	CustomEase.create('topas', 'M0,0 C0.2,0.8 0.2,1 1,1');

	const lenis = setupScroll();
	lenis.stop();
	initCursor();

	// 1. Le loader d'abord : rien de lourd ne dÃ©marre avant qu'il soit Ã  l'Ã©cran.
	await painted();

	// 2. Pendant le loader : 3D (shaders compilÃ©s, scÃ¨nes rendues une fois) et
	//    toutes les images (tÃ©lÃ©chargÃ©es et dÃ©codÃ©es). Plus rien Ã  charger ensuite.
	const glReady = loadGL();
	let onImg = () => {};
	const images = preloadAll((f) => onImg(f));
	const gl = await loadIntro({ glReady, images, onImages: (cb) => (onImg = cb) });
	if (gl) showGL(gl);
	else glReady.then((g) => showGL(g, true));

	// 3. Animations installÃ©es sous le loader (dÃ©coupes de texte, pinsâ€¦), puis ouverture.
	initSections(state, lenis);
	initMap({ gsap, ScrollTrigger });
	ScrollTrigger.refresh();
	await revealIntro(state);
	lenis.start();

	// Redimensionnement rÃ©el (largeur) : un seul recalcul, diffÃ©rÃ©.
	let lastW = innerWidth;
	let timer = 0;
	addEventListener('resize', () => {
		if (innerWidth === lastW) return;
		lastW = innerWidth;
		clearTimeout(timer);
		timer = setTimeout(() => ScrollTrigger.refresh(), 250);
	});
}

boot();
